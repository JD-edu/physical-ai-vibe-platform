cmake_minimum_required(VERSION 3.16)
project(PhyVibeBridge VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network Widgets)
add_executable(phyvibe-bridge src/main.cpp src/bridge_server.cpp src/bridge_server.h src/bridge_window.cpp src/bridge_window.h)
target_link_libraries(phyvibe-bridge PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Widgets)
target_compile_options(phyvibe-bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS phyvibe-bridge RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME bridge_network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_cpp_bridge.py $<TARGET_FILE:phyvibe-bridge>)
  add_test(NAME bridge_ui_lifecycle COMMAND phyvibe-bridge --smoke-test --port 0 --tcp-port 0 --host 127.0.0.1)
  set_tests_properties(bridge_ui_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
  set_tests_properties(bridge_network PROPERTIES TIMEOUT 40)
endif()
